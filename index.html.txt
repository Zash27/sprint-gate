  <script>
    let isPremium = false; 
    
    const COLOR_DIFF_THRESHOLD = 60;   
    const PIXEL_COUNT_THRESHOLD = 60;  
    
    const screens = { home: document.getElementById('homeScreen'), session: document.getElementById('sessionScreen'), history: document.getElementById('historyScreen') };
    const premiumModal = document.getElementById('premiumModal'), exitModal = document.getElementById('exitModal'), modalText = document.getElementById('modalText');
    const historyLock = document.getElementById('historyLock'), passesLock = document.getElementById('passesLock'), passesBlocker = document.getElementById('passesBlocker'), passesInput = document.getElementById('passesInput');
    const sessionNameLock = document.getElementById('sessionNameLock'), sessionNameBlocker = document.getElementById('sessionNameBlocker'), sessionNameInput = document.getElementById('sessionNameInput');
    const homeBadge = document.getElementById('homePremiumBadge'), sessionBadge = document.getElementById('sessionPremiumBadge');
    
    const video = document.getElementById('sourceVideo'), procCanvas = document.getElementById('processingCanvas'), procCtx = procCanvas.getContext('2d', { willReadFrequently: true });
    
    let cameraStream = null, animFrameId = null, audioCtx = null;
    let state = 'IDLE', startTime = 0, prevZoneData = null, countdownTimeout = null, isCancelled = false;
    let targetPasses = 1, currentPassCount = 0, sessionRunNumber = 1, gateCooldownEnd = 0;
    
    let currentRunId = null;
    let currentRunDomId = null;
    
    // Core Session Grouping Variables
    let currentSessionId = Date.now();
    let currentSessionDate = new Date().toLocaleString();

    const ZONE_PERCENT = 0.15;
    const ZONE_WIDTH = Math.round(procCanvas.width * ZONE_PERCENT);
    const ZONE_X = Math.floor((procCanvas.width - ZONE_WIDTH) / 2);

    function navTo(screenName) {
      if (screenName !== 'session' && cameraStream) stopCamera();
      Object.values(screens).forEach(s => s.classList.remove('active'));
      screens[screenName].classList.add('active');
      if (screenName === 'session') startCamera();
      else if (screenName === 'history') loadHistory();
    }

    function handleHistoryNav() { isPremium ? navTo('history') : showPremiumModal('Saved Device History'); }

    function handleSessionBack() {
      if (isPremium || sessionRunNumber === 1) navTo('home');
      else exitModal.style.display = 'flex';
    }

    function hideExitModal() { exitModal.style.display = 'none'; }

    function forceExit() {
      hideExitModal();
      sessionRunNumber = 1;
      document.getElementById('sessionLog').innerHTML = ''; 
      resetGate();
      navTo('home');
    }

    function showPremiumModal(featureName) {
      modalText.textContent = `Unlock Pro to access ${featureName}, remove ads, and track unlimited workouts permanently.`;
      premiumModal.style.display = 'flex';
    }
    function hidePremiumModal() { premiumModal.style.display = 'none'; }

    function togglePremiumTest() {
      isPremium = !isPremium;
      updatePremiumUI();
      alert(`Developer Mode: Premium is now ${isPremium ? 'ON' : 'OFF'}`);
    }

    function mockPurchase() {
      isPremium = true;
      updatePremiumUI();
      hidePremiumModal(); hideExitModal();
      alert("Purchase Successful! Pro Unlocked.");
    }

    function updatePremiumUI() {
      if (isPremium) {
        historyLock.style.display = 'none'; passesLock.style.display = 'none'; passesBlocker.style.display = 'none'; passesInput.removeAttribute('readonly');
        sessionNameLock.style.display = 'none'; sessionNameBlocker.style.display = 'none'; sessionNameInput.removeAttribute('readonly');
        homeBadge.style.display = 'block'; sessionBadge.style.display = 'block';
      } else {
        historyLock.style.display = 'inline'; passesLock.style.display = 'inline'; passesBlocker.style.display = 'block'; passesInput.setAttribute('readonly', true); passesInput.value = 1; 
        sessionNameLock.style.display = 'inline'; sessionNameBlocker.style.display = 'block'; sessionNameInput.setAttribute('readonly', true);
        homeBadge.style.display = 'none'; sessionBadge.style.display = 'none';
      }
    }
    updatePremiumUI();

    function attemptDeleteRun(runId, domId) {
      if (!isPremium) {
        showPremiumModal('Delete Runs');
        return;
      }
      const el = document.getElementById(domId);
      if (el) el.remove();
      
      let runs = JSON.parse(localStorage.getItem('sprintHistory') || '[]');
      runs = runs.filter(r => r.id !== runId);
      localStorage.setItem('sprintHistory', JSON.stringify(runs));
    }
    
    function deleteSession(sessionId) {
      if (confirm("Are you sure you want to delete this entire session and all its runs?")) {
        let runs = JSON.parse(localStorage.getItem('sprintHistory') || '[]');
        // Filter out all runs that share this session ID
        runs = runs.filter(r => (r.sessionId || r.date) != sessionId);
        localStorage.setItem('sprintHistory', JSON.stringify(runs));
        loadHistory(); 
      }
    }

    function saveRunToDevice(passes, finalTime, runNum, runId) {
      if (!isPremium) return; 
      const sName = sessionNameInput.value || 'Running Session';
      const runs = JSON.parse(localStorage.getItem('sprintHistory') || '[]');
      
      runs.unshift({ 
        id: runId, 
        sessionId: currentSessionId,
        sessionName: sName,
        sessionDate: currentSessionDate,
        passes, 
        time: finalTime, 
        runNumber: runNum 
      }); 
      
      localStorage.setItem('sprintHistory', JSON.stringify(runs));
    }

    function loadHistory() {
      const container = document.getElementById('historyContainer');
      const runs = JSON.parse(localStorage.getItem('sprintHistory') || '[]');
      if (runs.length === 0) { container.innerHTML = '<div id="noHistoryMsg">No runs saved yet. Go sprint!</div>'; return; }

      const sessionsMap = new Map();
      runs.forEach(run => {
        const sId = run.sessionId || run.date;
        if (!sessionsMap.has(sId)) {
          sessionsMap.set(sId, {
            sessionName: run.sessionName || 'Running Session',
            sessionDate: run.sessionDate || run.date,
            runs: []
          });
        }
        sessionsMap.get(sId).runs.push(run);
      });

      let html = '';
      sessionsMap.forEach((sessionData, sId) => {
        let runsHtml = sessionData.runs.map(r => `
          <div class="history-run-item">
            <span class="run-num">Run ${r.runNumber || 1}</span>
            <span class="run-passes">${r.passes} Pass${r.passes > 1 ? 'es' : ''}</span>
            <span class="run-time">${parseFloat(r.time).toFixed(3)}s</span>
          </div>
        `).join('');

        html += `
          <div class="history-card">
            <div class="history-card-header">
              <div class="history-session-info">
                <div class="history-session-name">${sessionData.sessionName}</div>
                <div class="history-date">${sessionData.sessionDate}</div>
              </div>
              <button class="delete-session-btn" onclick="deleteSession('${sId}')">🗑️</button>
            </div>
            <div class="history-runs-list">
              ${runsHtml}
            </div>
          </div>
        `;
      });

      container.innerHTML = html;
    }

    function startCamera() {
      navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment', width: { ideal: 640 }, height: { ideal: 480 } }, audio: false })
      .then(stream => { cameraStream = stream; video.srcObject = stream; video.play(); resetGate(); animFrameId = requestAnimationFrame(mainLoop); })
      .catch(err => alert("Camera Error: " + err.message));
    }

    function stopCamera() {
      if (cameraStream) { cameraStream.getTracks().forEach(track => track.stop()); cameraStream = null; }
      if (animFrameId) { cancelAnimationFrame(animFrameId); animFrameId = null; }
      video.srcObject = null;
    }

    function playBeep(freq, dur, type = 'sine') {
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        if (audioCtx.state === 'suspended') audioCtx.resume();
        const osc = audioCtx.createOscillator(), gain = audioCtx.createGain();
        osc.type = type; osc.frequency.value = freq;
        gain.gain.setValueAtTime(1.0, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
        osc.connect(gain); gain.connect(audioCtx.destination);
        osc.start(); osc.stop(audioCtx.currentTime + dur);
      } catch (e) {}
    }

    function logSessionUI(textLeft, textRight, cssClass) {
      if (!currentRunDomId) return;
      const runBlock = document.getElementById(currentRunDomId);
      if (!runBlock) return;
      const div = document.createElement('div');
      div.className = `log-item ${cssClass}`;
      div.innerHTML = `<span>${textLeft}</span> <span>${textRight}</span>`;
      runBlock.appendChild(div);
    }

    function sleep(ms) { return new Promise(r => { countdownTimeout = setTimeout(r, ms); }); }

    function resetGate() {
      isCancelled = true;
      clearTimeout(countdownTimeout);
      
      if (state !== 'FINISHED' && state !== 'IDLE' && currentRunDomId) {
        const el = document.getElementById(currentRunDomId);
        if (el) el.remove();
      }

      state = 'IDLE'; prevZoneData = null; currentPassCount = 0; gateCooldownEnd = 0;
      
      document.getElementById('statusBadge').textContent = 'READY'; document.getElementById('statusBadge').style.color = '#fff';
      document.getElementById('timerDisplay').textContent = '0.000s'; document.getElementById('timerDisplay').style.color = '#fff';
      document.getElementById('gateOverlay').className = '';

      document.getElementById('startBtn').disabled = false;
      document.getElementById('startBtn').textContent = sessionRunNumber > 1 ? 'NEXT RUN' : 'START';
      document.getElementById('delayInput').disabled = false;
    }

    function mainLoop() {
      animFrameId = requestAnimationFrame(mainLoop);
      if(video.videoWidth === 0) return; 

      procCtx.drawImage(video, 0, 0, procCanvas.width, procCanvas.height);
      const currentZoneData = procCtx.getImageData(ZONE_X, 0, ZONE_WIDTH, procCanvas.height);

      if (state === 'RUNNING') {
        const elapsed = (performance.now() - startTime) / 1000;
        document.getElementById('timerDisplay').textContent = elapsed.toFixed(2) + 's';

        if (elapsed > 1.0) {
          if (elapsed < gateCooldownEnd) { prevZoneData = new Uint8ClampedArray(currentZoneData.data); return; }
          if (prevZoneData) {
            let changedPixels = 0;
            const currentPixels = currentZoneData.data;
            for (let i = 0; i < currentPixels.length; i += 16) {
              const diff = Math.abs(currentPixels[i] - prevZoneData[i]) + Math.abs(currentPixels[i+1] - prevZoneData[i+1]) + Math.abs(currentPixels[i+2] - prevZoneData[i+2]);
              if (diff > COLOR_DIFF_THRESHOLD) changedPixels++;
            }
            if (changedPixels > PIXEL_COUNT_THRESHOLD) handleGateCross(elapsed);
          }
          prevZoneData = new Uint8ClampedArray(currentZoneData.data);
        }
      }
    }

    function handleGateCross(elapsed) {
      currentPassCount++;
      const overlay = document.getElementById('gateOverlay');

      if (currentPassCount < targetPasses) {
        playBeep(800, 0.2, 'square');
        overlay.className = 'triggered-lap';
        setTimeout(() => overlay.className = '', 400);
        logSessionUI(`Pass ${currentPassCount}`, `${elapsed.toFixed(3)}s`, 'log-lap');
        gateCooldownEnd = elapsed + 1.5; 
      } else {
        state = 'FINISHED';
        playBeep(1200, 0.6, 'triangle'); 
        document.getElementById('statusBadge').textContent = 'FINISHED!'; document.getElementById('statusBadge').style.color = '#34c759';
        document.getElementById('timerDisplay').textContent = elapsed.toFixed(3) + 's'; document.getElementById('timerDisplay').style.color = '#34c759';
        overlay.className = 'triggered-finish';

        logSessionUI(`FINAL TIME`, `${elapsed.toFixed(3)}s`, 'log-final');
        
        const delBtn = document.getElementById(`del-${currentRunId}`);
        if(delBtn) delBtn.style.display = 'block';

        saveRunToDevice(targetPasses, elapsed.toFixed(3), sessionRunNumber, currentRunId);
        
        sessionRunNumber++;
        document.getElementById('startBtn').disabled = false;
        document.getElementById('startBtn').textContent = 'NEXT RUN';
      }
    }

    async function startSequence() {
      isCancelled = false;
      if (!document.fullscreenElement && document.documentElement.requestFullscreen) document.documentElement.requestFullscreen().catch(()=>{});

      if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      if (audioCtx.state === 'suspended') await audioCtx.resume();

      document.getElementById('startBtn').disabled = true; document.getElementById('delayInput').disabled = true;
      targetPasses = isPremium ? (parseInt(passesInput.value, 10) || 1) : 1;
      currentPassCount = 0; gateCooldownEnd = 0;
      
      document.getElementById('gateOverlay').className = '';
      document.getElementById('timerDisplay').style.color = '#ffffff';

      currentRunId = Date.now();
      currentRunDomId = `run-${currentRunId}`;
      const runBlock = document.createElement('div');
      runBlock.className = 'run-block';
      runBlock.id = currentRunDomId;
      
      const delBtn = document.createElement('button');
      delBtn.className = 'delete-run-btn';
      delBtn.id = `del-${currentRunId}`;
      delBtn.innerHTML = '🗑️';
      const capturedId = currentRunId;
      const capturedDomId = currentRunDomId;
      delBtn.onclick = () => attemptDeleteRun(capturedId, capturedDomId);
      
      runBlock.appendChild(delBtn);
      document.getElementById('sessionLog').prepend(runBlock);

      logSessionUI(`Run ${sessionRunNumber}`, `${targetPasses} Pass(es)`, 'log-header');

      const delaySeconds = parseInt(document.getElementById('delayInput').value, 10) || 10;
      state = 'DELAY';

      for (let i = delaySeconds; i > 0; i--) {
        if (isCancelled) return;
        document.getElementById('statusBadge').textContent = 'WALK TO START'; document.getElementById('statusBadge').style.color = '#ff9500';
        document.getElementById('timerDisplay').textContent = `${i}s`;
        await sleep(1000);
      }

      state = 'COUNTDOWN';
      for (let count = 3; count >= 1; count--) {
        if (isCancelled) return;
        document.getElementById('statusBadge').textContent = 'GET SET'; document.getElementById('statusBadge').style.color = '#ffcc00';
        document.getElementById('timerDisplay').textContent = count;
        playBeep(520, 0.25);
        await sleep(1000);
      }

      if (isCancelled) return;
      document.getElementById('statusBadge').textContent = 'SPRINT!'; document.getElementById('statusBadge').style.color = '#34c759';
      document.getElementById('timerDisplay').textContent = 'GO!';
      playBeep(1040, 0.5);

      state = 'RUNNING'; startTime = performance.now(); prevZoneData = null;
    }

    document.getElementById('startBtn').addEventListener('click', () => { if (state === 'IDLE' || state === 'FINISHED') startSequence(); });
    document.getElementById('resetBtn').addEventListener('click', resetGate);
    
    document.getElementById('newSessionBtn').addEventListener('click', () => {
      sessionRunNumber = 1;
      currentSessionId = Date.now();
      currentSessionDate = new Date().toLocaleString();
      document.getElementById('sessionLog').innerHTML = ''; 
      sessionNameInput.value = 'Running Session';
      resetGate();
    });
